cmake_minimum_required(VERSION 3.22.1)
project(llamajni CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../native/llama.cpp)
if(NOT EXISTS "${LLAMA_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Restore the pinned llama.cpp submodule first; see llama-cpp/README.md")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_VULKAN_CHECK_RESULTS OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN_DEBUG OFF CACHE BOOL "" FORCE)
# Android uses the NDK's target loader, but shaders are built on the host.
# Optional paths support a user-local SDK without embedding machine paths.
if(DEFINED ENV{LLAMA_GLSLC})
    set(Vulkan_GLSLC_EXECUTABLE "$ENV{LLAMA_GLSLC}" CACHE FILEPATH "Host glslc" FORCE)
endif()
if(DEFINED ENV{LLAMA_VULKAN_INCLUDE_DIR})
    set(Vulkan_INCLUDE_DIR "$ENV{LLAMA_VULKAN_INCLUDE_DIR}" CACHE PATH "Vulkan headers" FORCE)
endif()
if(DEFINED ENV{LLAMA_SPIRV_HEADERS_DIR})
    set(SPIRV-Headers_DIR "$ENV{LLAMA_SPIRV_HEADERS_DIR}" CACHE PATH "SPIRV-Headers config" FORCE)
endif()
# This pin needs SPIRV-Headers as well as Vulkan-Hpp; it contains no bundled copy.
find_package(SPIRV-Headers CONFIG REQUIRED)
if(CMAKE_CROSSCOMPILING AND CMAKE_MAKE_PROGRAM AND
   (NOT GGML_VULKAN_SHADERS_GEN_TOOLCHAIN OR
    "${GGML_VULKAN_SHADERS_GEN_TOOLCHAIN}" STREQUAL "${CMAKE_CURRENT_BINARY_DIR}/vulkan-host-toolchain.cmake"))
    find_program(HOST_C_COMPILER NAMES gcc clang NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_program(HOST_CXX_COMPILER NAMES g++ clang++ NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    set(GGML_VULKAN_SHADERS_GEN_TOOLCHAIN "${CMAKE_CURRENT_BINARY_DIR}/vulkan-host-toolchain.cmake"
        CACHE FILEPATH "Host toolchain for Vulkan shader generation" FORCE)
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/vulkan-host-toolchain.cmake.in" "${GGML_VULKAN_SHADERS_GEN_TOOLCHAIN}" @ONLY)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/llama.cpp)
target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
if(ANDROID)
    include(vulkan-android.cmake)
endif()
add_library(llamajni SHARED llama_jni.cpp)
target_include_directories(llamajni PRIVATE
    ${LLAMA_ROOT}/include ${LLAMA_ROOT}/ggml/include
    ${LLAMA_ROOT}/common ${LLAMA_ROOT}/tools/mtmd ${LLAMA_ROOT}/vendor)
target_link_libraries(llamajni PRIVATE llama llama-common mtmd android log)
